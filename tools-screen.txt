class _PegonConverterScreen extends StatefulWidget {
  const _PegonConverterScreen();

  @override
  State<_PegonConverterScreen> createState() => _PegonConverterScreenState();
}

class _PegonConverterScreenState extends State<_PegonConverterScreen> {
  final _controller = TextEditingController();
  String _output = '';

  @override
  void dispose() {
    _controller.dispose();
    super.dispose();
  }

  Future<void> _convert() async {
    final result = await ApiService.instance.convertToPegon(_controller.text);
    setState(() => _output = result);
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Konverter Arabic Pegon')),
      body: ListView(
        padding: const EdgeInsets.all(16),
        children: [
          TextField(
            controller: _controller,
            maxLines: 4,
            decoration: const InputDecoration(
              labelText: 'Teks Latin',
              hintText: 'Ketik teks latin di sini...',
            ),
            onChanged: (_) => _convert(),
          ),
          const SizedBox(height: 16),
          Card(
            child: Padding(
              padding: const EdgeInsets.all(16),
              child: Column(
                crossAxisAlignment: CrossAxisAlignment.stretch,
                children: [
                  const Text(
                    'Hasil Pegon',
                    style: TextStyle(fontWeight: FontWeight.w600),
                  ),
                  const SizedBox(height: 8),
                  Text(
                    _output.isEmpty ? '...' : _output,
                    textAlign: TextAlign.right,
                    textDirection: TextDirection.rtl,
                    style: const TextStyle(fontSize: 22, height: 1.8),
                  ),
                ],
              ),
            ),
          ),
        ],
      ),
    );
  }
}

class _HijriCalendarScreen extends StatelessWidget {
  const _HijriCalendarScreen();

  @override

    final now = DateTime.now();
    const hijriMonths = [
      'Muharram', 'Safar', 'Rabiul Awal', 'Rabiul Akhir',

      'Ramadhan', 'Syawal', "Dzulqa'dah", 'Dzulhijjah',

      body: ListView(

                    style: const TextStyle(

                      fontWeight: FontWeight.bold,
                      color: AppConstants.emeraldDark,
                    ),
                  ),
                  const SizedBox(height: 4),
                  Text(
                    '${now.day}/${now.month}/${now.year} M',
                    style: const TextStyle(color: AppConstants.muted),
                  ),
                ],
              ),
            ),
          ),
          const SizedBox(height: 16),
          Card(
            child: Padding(
              padding: const EdgeInsets.all(16),
              child: Column(
                crossAxisAlignment: CrossAxisAlignment.start,
                children: [
                  const Text(
                    'Bulan Hijriyah',
                    style: TextStyle(
                      fontWeight: FontWeight.bold,
                      color: AppConstants.emeraldDark,
                    ),
                  ),
                  const SizedBox(height: 8),
                  ...hijriMonths.asMap().entries.map(
                        (e) => ListTile(
                          dense: true,
                          leading: CircleAvatar(
                            radius: 14,
                            backgroundColor: e.value == hijriMonth
                                ? AppConstants.emerald
                                : AppConstants.emerald.withValues(alpha: 0.12),
                            child: Text(
                              '${e.key + 1}',
                              style: TextStyle(
                                fontSize: 11,
                                color: e.value == hijriMonth
                                    ? Colors.white
                                    : AppConstants.emeraldDark,
                              ),
                            ),
                          ),
                          title: Text(e.value),
                        ),
                      ),
                ],
              ),
            ),
          ),
        ],
      ),
    );
  }
}

class _TimerScreen extends StatefulWidget {
  const _TimerScreen();

  @override
  State<_TimerScreen> createState() => _TimerScreenState();
}

class _TimerScreenState extends State<_TimerScreen> {
  int _seconds = 300;
  bool _running = false;

  @override
  Widget build(BuildContext context) {
    final m = (_seconds ~/ 60).toString().padLeft(2, '0');
    final s = (_seconds % 60).toString().padLeft(2, '0');

    return Scaffold(
      appBar: AppBar(title: const Text('Timer & Stopwatch')),
      body: Center(
        child: Column(
          mainAxisAlignment: MainAxisAlignment.center,
          children: [
            Text(
              '$m:$s',
              style: const TextStyle(
                fontSize: 72,
                fontWeight: FontWeight.bold,
                color: AppConstants.emeraldDark,
              ),
            ),
            const SizedBox(height: 24),
            Row(
              mainAxisAlignment: MainAxisAlignment.center,
              children: [
                FilledButton.icon(
                  onPressed: () => setState(() => _running = !_running),
                  icon: Icon(_running ? Icons.pause : Icons.play_arrow),
                  label: Text(_running ? 'Pause' : 'Start'),
                ),
                const SizedBox(width: 12),
                OutlinedButton.icon(
                  onPressed: () => setState(() {
                    _running = false;
                    _seconds = 300;
                  }),
                  icon: const Icon(Icons.refresh),
                  label: const Text('Reset'),
                ),
              ],
            ),
            const SizedBox(height: 24),
            Wrap(
              spacing: 8,
              children: [60, 300, 600, 900]
                  .map(
                    (sec) => ActionChip(
                      label: Text('${sec ~/ 60} menit'),
                      onPressed: () => setState(() {
                        _seconds = sec;
                        _running = false;
                      }),
                    ),
                  )
                  .toList(),
            ),
          ],
        ),
      ),
    );
  }
}
