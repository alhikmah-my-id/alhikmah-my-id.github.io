import 'package:flutter/material.dart';
import 'package:provider/provider.dart';
import 'package:share_plus/share_plus.dart';
import 'package:url_launcher/url_launcher.dart';

import '../../core/constants/app_constants.dart';
import '../../data/models/article_model.dart';
import '../../data/models/bookmark_model.dart';
import '../../data/providers/bookmark_provider.dart';
import '../widgets/tappx_banner.dart';

class ArticleDetailScreen extends StatelessWidget {
  final ArticleModel article;
  const ArticleDetailScreen({super.key, required this.article});

  @override
  Widget build(BuildContext context) {
    final bookmarks = context.watch<BookmarkProvider>();
    final isSaved = bookmarks.isBookmarked(article.id);

    return Scaffold(
      appBar: AppBar(
        title: Text(article.category),
        actions: [
          IconButton(
            icon: Icon(isSaved ? Icons.bookmark : Icons.bookmark_border),
            onPressed: () => bookmarks.toggle(
              BookmarkModel(
                articleId: article.id,
                title: article.title,
                slug: article.slug,
                category: article.category,
                savedAt: DateTime.now(),
              ),
            ),
          ),
          IconButton(
            icon: const Icon(Icons.share),
            onPressed: () => _share(context),
          ),
        ],
      ),
      body: ListView(
        padding: const EdgeInsets.fromLTRB(16, 16, 16, 32),
        children: [
          Text(
            article.title,
            style: const TextStyle(
              fontSize: 24,
              fontWeight: FontWeight.bold,
              height: 1.3,
            ),
          ),
          const SizedBox(height: 10),
          Row(
            children: [
              const Icon(Icons.person, size: 16, color: AppConstants.muted),
              const SizedBox(width: 6),
              Text(article.author, style: const TextStyle(fontSize: 13)),
              const Spacer(),
              const Icon(Icons.schedule, size: 16, color: AppConstants.muted),
              const SizedBox(width: 6),
              Text(article.readTime, style: const TextStyle(fontSize: 13)),
            ],
          ),
          const SizedBox(height: 16),
          const TappxBanner(),
          const SizedBox(height: 20),
          Text(
            article.excerpt,
            style: const TextStyle(fontSize: 15, height: 1.7),
          ),
          const SizedBox(height: 20),
          _section('Pendahuluan',
              'Pembahasan lengkap mengenai ${article.title}.'),
          _section('Dalil & Rujukan',
              "Berdasarkan Al-Qur'an, Hadits, dan pendapat ulama."),
          _section('Penerapan Praktis',
              'Langkah amaliyah dalam kehidupan sehari-hari.'),
          _section('Kesimpulan',
              'Semoga bermanfaat dan menjadi amal jariyah.'),
          const SizedBox(height: 24),
          _shareRow(context),
        ],
      ),
    );
  }

  Widget _section(String title, String body) {
    return Padding(
      padding: const EdgeInsets.only(bottom: 18),
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          Text(
            title,
            style: const TextStyle(
              fontSize: 18,
              fontWeight: FontWeight.bold,
              color: AppConstants.emeraldDark,
            ),
          ),
          const SizedBox(height: 8),
          Text(body, style: const TextStyle(fontSize: 15, height: 1.7)),
        ],
      ),
    );
  }

  Widget _shareRow(BuildContext context) {
    final url = '${AppConstants.siteUrl}/artikel/${article.slug}';
    return Wrap(
      spacing: 8,
      runSpacing: 8,
      children: [
        ActionChip(
          avatar: const Icon(Icons.chat, size: 16, color: Colors.white),
          label: const Text('WhatsApp', style: TextStyle(color: Colors.white)),
          backgroundColor: const Color(0xFF16A34A),
          onPressed: () => _launch('https://wa.me/?text=${Uri.encodeComponent('${article.title} $url')}'),
        ),
        ActionChip(
          avatar: const Icon(Icons.facebook, size: 16, color: Colors.white),
          label: const Text('Facebook', style: TextStyle(color: Colors.white)),
          backgroundColor: const Color(0xFF2563EB),
          onPressed: () => _launch('https://www.facebook.com/sharer/sharer.php?u=$url'),
        ),
        ActionChip(
          avatar: const Icon(Icons.alternate_email, size: 16, color: Colors.white),
          label: const Text('Twitter/X', style: TextStyle(color: Colors.white)),
          backgroundColor: const Color(0xFF0F172A),
          onPressed: () => _launch('https://twitter.com/intent/tweet?url=$url'),
        ),
        ActionChip(
          avatar: const Icon(Icons.send, size: 16, color: Colors.white),
          label: const Text('Telegram', style: TextStyle(color: Colors.white)),
          backgroundColor: const Color(0xFF0EA5E9),
          onPressed: () => _launch('https://t.me/share/url?url=$url'),
        ),
        ActionChip(
          avatar: const Icon(Icons.link, size: 16),
          label: const Text('Salin Link'),
          onPressed: () => _share(context),
        ),
      ],
    );
  }

  void _share(BuildContext context) {
    final url = '${AppConstants.siteUrl}/artikel/${article.slug}';
    Share.share('${article.title}\n$url');
  }

  Future<void> _launch(String url) async {
    final uri = Uri.parse(url);
    if (await canLaunchUrl(uri)) {
      await launchUrl(uri, mode: LaunchMode.externalApplication);
    }
  }
}
